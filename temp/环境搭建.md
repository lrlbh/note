### 编译固件

~~~shell
cd ~/mpy/micropython/ports/esp32

# 编译 ESP32-S3 的标准固件（推荐：带 Octal SPIRAM 的版本）
#   常用几种写法，任选一种：

# 方式A：最常用 - 带 8MB Octal PSRAM（绝大多数 ESP32-S3 开发板都用这个）
make -j30 BOARD=ESP32_GENERIC_S3 BOARD_VARIANT=SPIRAM_OCT

# 方式B：普通 SPIRAM（Quad 模式，较少板子使用）
make BOARD=ESP32_GENERIC_S3

# 方式C：无外部 PSRAM（只用内部 512KB SRAM，性能最差，不推荐）
make BOARD=ESP32_GENERIC_S3 BOARD_VARIANT=NO_PSRAM   # 或干脆不写 BOARD_VARIANT

# 方式D：如果你想要 USB + CDC REPL + Octal PSRAM（新版支持更好）
make BOARD=ESP32_GENERIC_S3 BOARD_VARIANT=SPIRAM_OCT USB=1

# （可选）清理上一次编译结果，重新开始
make BOARD=ESP32_GENERIC_S3 BOARD_VARIANT=SPIRAM_OCT clean
~~~

~~~C
#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include "driver/spi_master.h"
#include "driver/spi_common.h"
#include <string.h>        // 为了 memset
#include <esp_heap_caps.h> // 如果想指定内存类型（推荐）
#include "esp_timer.h"
#include <esp_psram.h>
#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"

#define PIN_RST GPIO_NUM_14
#define PIN_BL GPIO_NUM_48
#define PIN_DC GPIO_NUM_21

#define LCD_RST(x) gpio_set_level(PIN_RST, (x))
#define LCD_BLK(x) gpio_set_level(PIN_BL, (x))
#define LCD_DC(x) gpio_set_level(PIN_DC, (x))
// #define LCD_CS(x) x ? gpio_set_level(GPIO_NUM_47, 1) : gpio_set_level(GPIO_NUM_47, 0)

spi_device_handle_t spi2;

#define LCD_WIDTH 320
#define LCD_HEIGHT 480

// UINT8指针？
void send_datas(uint8_t *data, uint32_t len)
{

    LCD_DC(1);

    static spi_transaction_t t = {0};
    t.tx_buffer = data;

    while (1)
    {
        if (len > 32768)
        {
            t.length = 262144;
            t.tx_buffer = (uint8_t *)tx_buffer + 32768;
            spi_device_polling_transmit(spi2, &t);
        }
        else
        {
            if (len <= 0)
                return;
            t.length = len * 8;
            spi_device_polling_transmit(spi2, &t);
        }
    }
}

static inline void send_data(uint8_t data)
{
    // static
    LCD_DC(1);
    spi_transaction_t t = {0};
    t.length = 8;
    t.tx_data[0] = data;
    t.flags = SPI_TRANS_USE_TXDATA;
    spi_device_polling_transmit(spi2, &t);
}

static inline void send_cmd(uint8_t data)
{
    LCD_DC(0);
    spi_transaction_t t = {0};
    t.length = 8;
    t.tx_data[0] = data;
    t.flags = SPI_TRANS_USE_TXDATA;
    spi_device_polling_transmit(spi2, &t);
}

static inline void lcd_clear(void *data, uint32_t len)
{

    spi_transaction_t t = {0};
    t.tx_buffer = data;
    // t.flags = SPI_TRANS_USE_TXDATA;

    while (1)
    {
        if (len > 32768)
        {
            len -= 32768;
            t.length = 262144;
            // t.tx_buffer = data;
            spi_device_polling_transmit(spi2, &t);
        }
        else
        {
            t.length = len * 8;
            // printf("%d\n", t.length);
            // t.tx_buffer = data;
            spi_device_polling_transmit(spi2, &t);
            break;
        }
    }
}

void lcd_init(void)
{
    // 1. 硬件复位
    LCD_RST(0);
    vTaskDelay(pdMS_TO_TICKS(20));
    LCD_RST(1);
    vTaskDelay(pdMS_TO_TICKS(200)); // 等待复位完成

    // 2. 背光先开（方便观察白屏）
    LCD_BLK(1);

    // 3. Sleep Out
    send_cmd(0x11);
    vTaskDelay(pdMS_TO_TICKS(120));

    // 4. Memory Data Access Control
    send_cmd(0x36);
    send_data(0x48); // RGB竖屏模式，可根据需要修改

    // 5. Interface Pixel Format
    send_cmd(0x3A);
    send_data(0x05); // 16-bit/pixel

    // 6. Command Set Control (扩展指令)
    send_cmd(0xF0);
    send_data(0xC3); // 扩展模式A
    send_cmd(0xF0);
    send_data(0x96); // 扩展模式B

    // 7. Display Function Control
    send_cmd(0xB4);
    send_data(0x01); // 显示功能参数

    // 8. Gate Control
    send_cmd(0xB7);
    send_data(0xC6);

    // 9. VCOM Control
    send_cmd(0xB9);
    send_data(0x02); // VCOM电压
    send_data(0xE0); // VCOM偏移

    // 10. Power Control
    send_cmd(0xC0);
    send_data(0x00); // AVDD
    send_data(0x00); // VGH/VGL

    send_cmd(0xC1);
    send_data(0x1B); // 电源参数

    send_cmd(0xC2);
    send_data(0xA7);

    send_cmd(0xC5);
    send_data(0x00); // VCOM调节

    // 11. Driver Timing Control A
    send_cmd(0xE8);
    send_data(0x40);
    send_data(0x8A);
    send_data(0x00);
    send_data(0x00);
    send_data(0x33);
    send_data(0x19);
    send_data(0xA5);
    send_data(0x33);

    // 12. Gamma 校正
    send_cmd(0xE0); // Positive Gamma
    send_data(0xF0);
    send_data(0x09);
    send_data(0x0F);
    send_data(0x0D);
    send_data(0x0D);
    send_data(0x1C);
    send_data(0x3D);
    send_data(0x44);
    send_data(0x55);
    send_data(0x39);
    send_data(0x18);
    send_data(0x18);
    send_data(0x36);
    send_data(0x39);

    send_cmd(0xE1); // Negative Gamma
    send_data(0xF0);
    send_data(0x09);
    send_data(0x0F);
    send_data(0x09);
    send_data(0x08);
    send_data(0x01);
    send_data(0x31);
    send_data(0x33);
    send_data(0x46);
    send_data(0x09);
    send_data(0x13);
    send_data(0x13);
    send_data(0x2A);
    send_data(0x31);

    // 13. Exit Command Set Control
    send_cmd(0xF0);
    send_data(0x3C); // 回到标准模式A
    send_cmd(0xF0);
    send_data(0x69); // 回到标准模式B

    vTaskDelay(pdMS_TO_TICKS(120));

    // 14. Tearing Effect Line OFF
    send_cmd(0x35);
    send_data(0x00);

    // 15. Display Inversion ON
    send_cmd(0x21);

    // 16. Display ON
    send_cmd(0x29);
    vTaskDelay(pdMS_TO_TICKS(50));

    // 17. 设置列/行地址 (480x320)
    send_cmd(0x2A); // Column Address Set
    send_data(0x00);
    send_data(0x00);
    send_data(0x01);
    send_data(0x3F);

    send_cmd(0x2B); // Row Address Set
    send_data(0x00);
    send_data(0x00);
    send_data(0x01);
    send_data(0xDF);

    // 18. Memory Write 开始
    send_cmd(0x2C);

    // 19. 可选：清屏，填充黑色
    // lcd_clear(black_buf, 320*480); // black_buf = 预先分配的黑色缓冲区

    // Column Address Set (0x2A): 0 ~ 319
    send_cmd(0x2A);
    LCD_DC(1);
    send_data(0x00);
    send_data(0x00);
    send_data(0x01);
    send_data(0x3F); // 319 = 0x013F

    // Page Address Set (0x2B): 0 ~ 479
    send_cmd(0x2B);
    LCD_DC(1);
    send_data(0x00);
    send_data(0x00);
    send_data(0x01);
    send_data(0xDF); // 479 = 0x01DF

    // Memory Write 开始
    send_cmd(0x2C);
    LCD_DC(1);

    // static
    LCD_DC(1);
}

void LCD_INIT(void)
{

    // 初始化引脚
    gpio_config_t io = {
        .mode = GPIO_MODE_OUTPUT,
        .pin_bit_mask =
            (1ULL << PIN_RST) |
            (1ULL << PIN_BL) |
            (1ULL << PIN_DC),
    };
    gpio_config(&io);

    // 初始化SPI
    spi_bus_config_t spi_config = {

        .flags = SPICOMMON_BUSFLAG_MASTER,        // 检测那些错误
        .intr_flags = 0,                          // 硬件中断标志
        .isr_cpu_id = ESP_INTR_CPU_AFFINITY_AUTO, // 中断那个CPU
        .max_transfer_sz = 320 * 480 * 2,         // 最大传输字节
        .miso_io_num = -1,                        // 收
        .mosi_io_num = GPIO_NUM_13,               // 发
        .quadhd_io_num = -1,
        .quadwp_io_num = -1,
        .sclk_io_num = GPIO_NUM_12, // 时钟
    }; // dc = 21 cs = 47

    spi_bus_initialize(SPI2_HOST, &spi_config, SPI_DMA_CH_AUTO); // SPI_DMA_DISABLED

    spi_device_interface_config_t spi_dev = {
        .clock_source = SPI_CLK_SRC_DEFAULT, // 时钟源
        .clock_speed_hz = 80000000,          // 速度
        .mode = 0,                           // 极性相位
        .queue_size = 10,
        .spics_io_num = GPIO_NUM_47,    // cs
        .flags = SPI_DEVICE_HALFDUPLEX, // 只发不收
    };
    spi_bus_add_device(SPI2_HOST, &spi_dev, &spi2);

    lcd_init();
}
~~~
