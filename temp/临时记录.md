在/data目录下创建一个脚本，其他目录下创建重启后会被自动删除

~~~shell
#!/bin/sh
ip6tables -I forwarding_rule -d ::200/::ffff:ffff:ffff:ffff -j ACCEPT  # ffff掩码 被ffff包裹的为需要/前面匹配上才可以访问
ip6tables -I forwarding_rule -p tcp --dport 32400 -j ACCEPT  # 也可以指定端口和通信协议，所有端口都将被开放
~~~

将脚本添加到防火墙启动项，相当于开机启动，在最下方追加

~~~shell
# vim /data/etc/config/firewall
# /etc/init.d/firewall restart
config include 'start' # 应该只要不重复即可
        option type 'script'
        option path '/data/start_sh/start.sh' # 脚本所在目录
        option enabled '1'
~~~
